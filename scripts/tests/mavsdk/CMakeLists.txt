cmake_minimum_required(VERSION 3.20)
project(mavsdk_archive_verification LANGUAGES CXX)
find_package(MAVSDK 4.1.0 EXACT REQUIRED CONFIG)
find_package(Threads REQUIRED)
add_executable(mavsdk_udp_verification udp_verification.cpp)
target_compile_features(mavsdk_udp_verification PRIVATE cxx_std_20)
target_compile_options(mavsdk_udp_verification PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(mavsdk_udp_verification PRIVATE MAVSDK::mavsdk Threads::Threads)
enable_testing()
add_test(NAME mavsdk_udp_verification COMMAND mavsdk_udp_verification)
set_tests_properties(mavsdk_udp_verification PROPERTIES TIMEOUT 90)
